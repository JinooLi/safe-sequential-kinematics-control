cmake_minimum_required(VERSION 3.10)
project(panda_replay LANGUAGES CXX)
option(PANDA_WITH_FRANKA "Build hardware modes using the installed libfranka" OFF)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(panda_trajectory trajectory.cpp)
target_include_directories(panda_trajectory PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(panda_trajectory PRIVATE -Wall -Wextra -Wpedantic)
add_executable(panda_replay main.cpp)
target_link_libraries(panda_replay PRIVATE panda_trajectory)
target_compile_options(panda_replay PRIVATE -Wall -Wextra -Wpedantic)
if(PANDA_WITH_FRANKA)
  find_package(Franka 0.8 REQUIRED)
  target_compile_definitions(panda_replay PRIVATE HAVE_FRANKA)
  target_link_libraries(panda_replay PRIVATE Franka::Franka)
endif()
enable_testing()
add_executable(test_trajectory tests/test_trajectory.cpp)
target_link_libraries(test_trajectory PRIVATE panda_trajectory)
add_test(NAME trajectory COMMAND test_trajectory)
if(PANDA_WITH_FRANKA)
  add_executable(test_franka_filter tests/test_franka_filter.cpp)
  target_link_libraries(test_franka_filter PRIVATE panda_trajectory Franka::Franka)
  add_test(NAME franka_filter COMMAND test_franka_filter)
endif()
